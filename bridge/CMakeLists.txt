cmake_minimum_required(VERSION 3.20)
project(overlord_bridge LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(WIN32)
  add_library(overlord_bridge SHARED extension.cpp)
  target_link_libraries(overlord_bridge PRIVATE winhttp)
  set_target_properties(overlord_bridge PROPERTIES PREFIX "" OUTPUT_NAME "overlord_bridge_x64")
  if(MINGW)
    target_link_options(overlord_bridge PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
endif()
add_executable(queue_test ../tests/native/queue_test.cpp)
if(MINGW)
  target_link_options(queue_test PRIVATE -static -static-libgcc -static-libstdc++)
endif()
enable_testing()
add_test(NAME queue_bounds COMMAND queue_test)
if(WIN32)
  add_executable(extension_smoke ../tests/native/extension_smoke.cpp)
  if(MINGW)
    target_link_options(extension_smoke PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
endif()
